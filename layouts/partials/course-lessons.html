{{/* A course's outline, line by line, each matched to its episode if one exists. Input: the course page (a standing page whose title is the category name). Output: a slice of dicts, in outline order: n lesson number (position in the list, filmed or not) title the move's name as written (your class name) alt its traditional name, from a trailing parenthesis, or "" key the name, normalised (title-key.html) episode the episode page, or false if not filmed yet An outline line may carry a traditional name in parentheses at the end: "Energy ball right to ward off left (Grasp the Sparrow's Tail: Ward Off Left)" It's shown smaller beneath the name and plays no part in matching. The outline is the FIRST numbered list on the course page. A line finds its episode in one of two ways: 1. it's a link to a post (a link always wins), or 2. a post in the course's category has the same title, forgiving capitals and punctuation. If two posts match, the earliest wins, which is why a deep dive published a week later can never take an episode's place. Nothing checks for a video. An episode is whatever the outline points at, so changing video providers never changes what counts as a lesson. Expensive, so call it through partialCached keyed by the course page: {{ $lessons := partialCached "course-lessons.html" $course $course.RelPermalink }} */}} {{- $course := . -}} {{- $courseKey := partial "norm-title.html" $course.Title -}} {{- /* 1. the outline lines */ -}} {{- $items := slice -}} {{- with findRE `(?s)]*>.*?` (string $course.Content) 1 -}} {{- range findRESubmatch `(?s)]*>(.*?)` (index . 0) -}} {{- $inner := index . 1 -}} {{- $href := "" -}} {{- with findRESubmatch `href="([^"]+)"` $inner 1 }}{{ $href = htmlUnescape (index (index . 0) 1) }}{{ end -}} {{- $name := partial "split-name.html" ($inner | plainify | htmlUnescape) -}} {{- $items = $items | append (dict "text" $name.main "alt" $name.alt "key" (partial "norm-title.html" $name.main) "href" $href) -}} {{- end -}} {{- end -}} {{- /* 2. the course's posts by normalised title, earliest first */ -}} {{- $byTitle := dict -}} {{- range sort (partial "posts.html" .) "Date" "asc" -}} {{- $inCourse := false -}} {{- range .Params.categories }}{{ if eq (partial "norm-title.html" .) $courseKey }}{{ $inCourse = true }}{{ end }}{{ end -}} {{- if and $inCourse .Title -}} {{- $t := partial "title-key.html" .Title -}} {{- if not (index $byTitle $t) }}{{ $byTitle = merge $byTitle (dict $t .) }}{{ end -}} {{- end -}} {{- end -}} {{- /* 3. match */ -}} {{- $urls := partialCached "url-index.html" "urls" -}} {{- $lessons := slice -}} {{- range $i, $it := $items -}} {{- $ep := false -}} {{- with $it.href -}} {{- with partial "href-to-rel.html" . }}{{ with index $urls . }}{{ $ep = . }}{{ end }}{{ end -}} {{- end -}} {{- if not $ep }}{{ with index $byTitle $it.key }}{{ $ep = . }}{{ end }}{{ end -}} {{- $lessons = $lessons | append (dict "n" (add $i 1) "title" $it.text "alt" $it.alt "key" $it.key "episode" $ep) -}} {{- end -}} {{- return $lessons -}}
