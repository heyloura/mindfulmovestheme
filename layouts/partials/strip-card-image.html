. Usage: {{ $body := partial "strip-card-image.html" .Content }} */}} {{- $html := string . -}} {{- $html = replaceRE `

\s*` "" $html -}} {{- $html = replaceRE `]*class="[^"]*\bcard-image\b[^"]*"[^>]*>\s*` "" $html -}} {{- return ($html | safeHTML) -}}
